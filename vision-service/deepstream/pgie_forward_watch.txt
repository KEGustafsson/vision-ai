# nvinfer primary GIE config for the forward-watch marine obstacle model
# (YOLOv8n, 6 classes). Selected by detector.model=forward-watch in the YAML.
# Only ONE detection model runs at a time — this config replaces pgie_yolo11n.txt
# when forward-watch is selected; the two are never active together.
#
# MODEL SOURCE
# ============
# Fetch + convert in one step (recommended):
#   python3 scripts/download_forward_watch.py     # downloads AND converts
# The published ONNX
#   https://github.com/SkipperDon/signalk-forward-watch/blob/main/models/forward-watch.onnx
# is a STOCK Ultralytics export (output [1, 4+nc, 8400]). The custom parser below
# CANNOT read that layout — fed as-is it yields ZERO detections. It must be
# rewritten to the DeepStream-Yolo layout ([1, 8400, 6] = x1,y1,x2,y2,score,class)
# via scripts/convert_to_deepstream.py (the download script does this for you).
# Place the converted file in this directory as forward-watch.onnx.
#
# This model detects 6 marine-specific classes:
#   0=ship  1=boat  2=debris  3=buoy  4=kayak  5=log
#
# Same YOLOv8n architecture and 640x640 input as the COCO model; once converted,
# the same deepstream-yolo custom parser works unchanged — only the label file
# and num-detected-classes differ (6 vs 80).
#
# CUSTOM PARSER
# =============
# Uses the same libnvdsinfer_custom_impl_Yolo.so as the COCO model — build it
# once on the Jetson (see pgie_yolo11n.txt for the steps). The COCO ONNX is made
# parser-ready at build time by DeepStream-Yolo's export_yoloV8.py; the
# forward-watch ONNX is made parser-ready by scripts/convert_to_deepstream.py
# (both produce the identical [N,6] head this parser decodes).

[property]
gpu-id=0

# 0=RGB (YOLOv8 expects RGB, channels-first)
model-color-format=0

# 1/255 — match Ultralytics normalisation
net-scale-factor=0.0039215697906911373

onnx-file=forward-watch.onnx
# nvinfer auto-builds the TRT engine from the ONNX on first run (deepstream-yolo
# names it <onnx>_b<N>_gpu0_fp16) and writes it next to the ONNX.
model-engine-file=forward-watch.onnx_b2_gpu0_fp16.engine

labelfile-path=labels_forward_watch.txt

# Match the number of cameras in deepstream.yaml (nvstreammux batch).
batch-size=2

# Sole primary detector — must stay 1 so nvtracker correlates with it.
gie-unique-id=1

# 0=FP32  1=INT8  2=FP16 (FP16 is the Jetson Orin sweet spot)
network-mode=2

# forward-watch has 6 output classes.
num-detected-classes=6

interval=0

# 0=detector  1=classifier  2=segmentation
network-type=0

# cluster-mode=2 = NMS, the mode the deepstream-yolo YOLOv8 parser is written for.
cluster-mode=2

parse-bbox-func-name=NvDsInferParseYolo
# The parser is named by ABSOLUTE path, NOT relative to this file: compose
# bind-mounts the host's deepstream/ over this directory, so a relative name
# would resolve to the HOST copy — which on a JetPack 5 board (DeepStream 6.3 /
# CUDA 11.4) is the wrong ABI and fails to load. /opt/vision-service/lib/ is
# outside the mount, so each image supplies the build matching its own
# DeepStream (see Dockerfile.deepstream and Dockerfile.deepstream.xavier).
custom-lib-path=/opt/vision-service/lib/libnvdsinfer_custom_impl_Yolo.so

# Preserve aspect ratio by letterboxing — prevents horizontal FoV distortion.
maintain-aspect-ratio=1
symmetric-padding=1

[class-attrs-all]
# Keep low here; the pipeline's stabilizer and confidence gate filter downstream.
pre-cluster-threshold=0.1
nms-iou-threshold=0.45
topk=300
