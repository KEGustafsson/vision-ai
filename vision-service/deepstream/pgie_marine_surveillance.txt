# nvinfer primary GIE config for the Roboflow Marine Surveillance model
# (YOLOv8s, 7 classes). Selected by detector.model=marine-surveillance in the YAML.
# Only ONE detection model runs at a time — this config replaces pgie_yolo11n.txt
# when marine-surveillance is selected; the models are never active together.
#
# MODEL SOURCE
# ============
# Trained ON-BOX from the Roboflow "Marine Surveillance" dataset (you do not get
# the author's weights — Roboflow only exports the dataset). Run:
#   python3 training/train_marine_surveillance.py --api-key <ROBOFLOW_KEY> \
#       --workspace <ws> --project <proj> --version <n>
# It downloads the dataset, trains YOLOv8s, does a stock ONNX export and runs it
# through scripts/convert_to_deepstream.py (the proven [N,6] rewrite, same one
# used for forward-watch), writing marine-surveillance.onnx here and regenerating
# labels_marine_surveillance.txt.
#
# 7 classes (kept DISTINCT, dataset/alphabetical order):
#   0=boat 1=buoy 2=kayak 3=sail-boat 4=speed-boat 5=vessel 6=war-ship
# NO person class — selecting this model disables man-overboard detection.
#
# CUSTOM PARSER
# =============
# Same libnvdsinfer_custom_impl_Yolo.so as the COCO/forward-watch models (build it
# once on the Jetson — see pgie_yolo11n.txt). The training script's stock export is
# rewritten to the [N,6] head this parser decodes by convert_to_deepstream.py.

[property]
gpu-id=0

# 0=RGB (YOLOv8 expects RGB, channels-first)
model-color-format=0

# 1/255 — match Ultralytics normalisation
net-scale-factor=0.0039215697906911373

onnx-file=marine-surveillance.onnx
# nvinfer auto-builds the TRT engine from the ONNX on first run (deepstream-yolo
# names it <onnx>_b<N>_gpu0_fp16) and writes it next to the ONNX.
model-engine-file=marine-surveillance.onnx_b2_gpu0_fp16.engine

labelfile-path=labels_marine_surveillance.txt

# Match the number of cameras in deepstream.yaml (nvstreammux batch).
batch-size=2

# Sole primary detector — must stay 1 so nvtracker correlates with it.
gie-unique-id=1

# 0=FP32  1=INT8  2=FP16 (FP16 is the Jetson Orin sweet spot)
network-mode=2

# Marine Surveillance has 7 output classes.
num-detected-classes=7

interval=0

# 0=detector  1=classifier  2=segmentation
network-type=0

# cluster-mode=2 = NMS, the mode the deepstream-yolo YOLOv8 parser is written for.
cluster-mode=2

parse-bbox-func-name=NvDsInferParseYolo
custom-lib-path=libnvdsinfer_custom_impl_Yolo.so

# Preserve aspect ratio by letterboxing — prevents horizontal FoV distortion.
maintain-aspect-ratio=1
symmetric-padding=1

[class-attrs-all]
# Keep low here; the pipeline's stabilizer and confidence gate filter downstream.
pre-cluster-threshold=0.1
nms-iou-threshold=0.45
topk=300
