# nvinfer primary GIE config for YOLOv8n (DeepStream 7.x / JetPack 6)
#
# ENGINE FORMAT REQUIREMENT
# =========================
# This config expects a YOLOv8n ONNX engine WITHOUT end-to-end NMS.
# The Ultralytics default export (export_engine.py) bakes NMS into the engine,
# which is incompatible with the deepstream-yolo custom parser below.
#
# To export the correct engine for DeepStream:
#   python -c "
#   from ultralytics import YOLO
#   YOLO('models/yolov8n.pt').export(
#       format='onnx', simplify=True, opset=17, imgsz=640)
#   "
#   # Then build TRT engine from the ONNX (DeepStream does this on first run
#   # if model-engine-file does not exist, using the engine-create-func):
#   cp models/yolov8n.onnx deepstream/yolov8n_raw.onnx
#   # OR use trtexec manually and set model-engine-file accordingly.
#
# CUSTOM PARSER (deepstream-yolo)
# ================================
# Build libnvdsinfer_custom_impl_Yolo.so on the Jetson once:
#   git clone https://github.com/marcoslucianops/DeepStream-Yolo /tmp/ds-yolo
#   cd /tmp/ds-yolo
#   CUDA_VER=$(nvcc --version | grep -oP 'V\K[0-9]+\.[0-9]+') \
#       make -C nvdsinfer_custom_impl_Yolo
#   cp nvdsinfer_custom_impl_Yolo/libnvdsinfer_custom_impl_Yolo.so \
#       /path/to/vision-ai/vision-service/deepstream/

[property]
gpu-id=0

# 1/255 — match Ultralytics normalisation
net-scale-factor=0.0039215697906911373

# Path relative to this config file's directory
model-engine-file=../models/yolov8n_ds.engine

# If the .engine file does not exist, nvinfer will build it from the ONNX.
# Uncomment and point to the exported ONNX if you want auto-build:
# onnx-file=yolov8n_raw.onnx

labelfile-path=labels_coco.txt

# Match number of cameras in deepstream.yaml
batch-size=2

# Unique ID used by nvtracker to correlate inference with stream metadata
gie-unique-id=1

# 0=FP32  1=INT8  2=FP16 (FP16 is the Jetson Orin sweet spot)
network-mode=2

# Number of output classes (80 COCO; extend if using a fine-tuned marine model)
num-detected-classes=80

# Run inference on every frame (interval=0); set to 1 to infer every 2nd frame
interval=0

# Zero mean subtraction (YOLOv8 uses only scale, no per-channel means)
gie-std-means=0;0;0;0

# CHW format, 640x640 input
input-dims=3;640;640;0

# 0=detector  1=classifier  2=segmentation
network-type=0

# cluster-mode=4 uses NMS internally — correct for raw YOLOv8 output (no built-in NMS)
cluster-mode=4

# deepstream-yolo parser for YOLOv8 detection output format
parse-bbox-func-name=NvDsInferParseCustomYoloV8
custom-lib-path=libnvdsinfer_custom_impl_Yolo.so

# Output tensor name from the ONNX export (check with: polygraphy inspect model yolov8n.onnx)
output-blob-names=output0

# Preserve aspect ratio by letterboxing — prevents horizontal FoV distortion
maintain-aspect-ratio=1
symmetric-padding=1

[class-attrs-all]
# Minimum confidence for nvinfer to pass a bbox to nvtracker.
# Keep low here; the pipeline's stabilizer and confidence gate filter downstream.
pre-cluster-threshold=0.1
nms-iou-threshold=0.45
