# nvinfer primary GIE config for YOLO11n.
# Shared by both DeepStream images — JetPack 6 / DS 7.1 (Dockerfile.deepstream)
# and JetPack 5 / DS 6.3 (Dockerfile.deepstream.xavier). Every key below is
# version-neutral; the one platform-dependent thing, the custom parser, is
# referenced by absolute path so each image supplies its own build (see the
# note at custom-lib-path).
#
# ENGINE FORMAT REQUIREMENT
# =========================
# This config expects a YOLO11n ONNX WITHOUT end-to-end NMS, exported with the
# marcoslucianops DeepStream-Yolo export script (NOT the Ultralytics default
# export, which bakes NMS in and is incompatible with the custom parser below).
# Dockerfile.deepstream does this automatically; to reproduce by hand:
#   git clone https://github.com/marcoslucianops/DeepStream-Yolo /tmp/ds-yolo
#   cp /tmp/ds-yolo/utils/export_yolo11.py .
#   python3 export_yolo11.py -w yolo11n.pt -s 768 --opset 16 --dynamic --simplify
#   mv yolo11n.onnx deepstream/yolo11n_ds.onnx
#   # nvinfer then builds the TRT engine from the ONNX on first run (if
#   # model-engine-file does not exist), or use trtexec and set it accordingly.
#   # -s 768 MUST match detector.imgsz in deepstream.yaml (drives tracker_width).
#
# CUSTOM PARSER (deepstream-yolo)
# ================================
# Build libnvdsinfer_custom_impl_Yolo.so on the Jetson once:
#   git clone https://github.com/marcoslucianops/DeepStream-Yolo /tmp/ds-yolo
#   cd /tmp/ds-yolo
#   CUDA_VER=$(nvcc --version | grep -oP 'V\K[0-9]+\.[0-9]+') \
#       make -C nvdsinfer_custom_impl_Yolo
#   cp nvdsinfer_custom_impl_Yolo/libnvdsinfer_custom_impl_Yolo.so \
#       /path/to/vision-ai/vision-service/deepstream/

# NOTE: the keys below mirror the official deepstream-yolo YOLO11 sample
# (config_infer_primary_yolo11.txt) so they stay in lockstep with the parser
# (.so) and the ONNX that Dockerfile.deepstream builds from the SAME repo.
# YOLO11 shares YOLOv8's output layout, so the SAME NvDsInferParseYolo parser
# (.so) and COCO labels apply — only the ONNX weights change.
[property]
gpu-id=0

# 0=RGB (YOLOv8 expects RGB, channels-first)
model-color-format=0

# 1/255 — match Ultralytics normalisation
net-scale-factor=0.0039215697906911373

# ONNX (baked in this dir) → nvinfer auto-builds the TRT engine on first run and
# writes it to ../models (bind-mounted, so the engine persists across recreates).
# Both paths are relative to THIS config file's directory.
onnx-file=yolo11n_ds.onnx
# Prebuilt engine baked into this dir (deepstream-yolo names it <onnx>_b<N>_gpu0_fp16
# when nvinfer builds from the ONNX). Deserialised on startup — no ~7.5-min rebuild.
# Delete it (and rebuild the image) to force a fresh build, e.g. after a TRT/HW change.
#
# CROSS-BOARD NOTE: this filename is platform-NEUTRAL while the engine itself is
# not — TensorRT engines are tied to the GPU arch and TRT version, and the host's
# deepstream/ dir is bind-mounted, so moving a checkout between an Orin (TRT 10)
# and a Xavier (TRT 8.5) leaves a stale engine here that this board cannot load.
# That is SAFE, not fatal: nvinfer falls back to rebuilding from onnx-file.
# Observed verbatim on the Xavier's first start:
#   deserialize backend context from engine from file :.../yolo11n_ds.onnx_b2_gpu0_fp16.engine failed, try rebuild
#   Info from NvDsInferContextImpl::buildModel(): Trying to create engine from model files
# The only cost is the rebuild (~19 min on a Xavier NX), so the engine is not
# given a per-platform name — that would need per-platform pgie configs, and the
# single shared set of configs is worth more than avoiding a rare one-off wait.
# Delete the stale *_gpu0_fp16.engine yourself if you want to skip the fallback.
model-engine-file=yolo11n_ds.onnx_b2_gpu0_fp16.engine

labelfile-path=labels_coco.txt

# Match the number of cameras in deepstream.yaml (nvstreammux batch).
batch-size=2

# Unique ID used by nvtracker to correlate inference with stream metadata
gie-unique-id=1

# 0=FP32  1=INT8  2=FP16 (FP16 is the Jetson Orin sweet spot)
network-mode=2

# Number of output classes (80 COCO; extend if using a fine-tuned marine model)
num-detected-classes=80

# Run inference on every frame (interval=0); set to 1 to infer every 2nd frame
interval=0

# 0=detector  1=classifier  2=segmentation
network-type=0

# cluster-mode=2 = NMS, the mode the deepstream-yolo YOLOv8 parser is written for.
cluster-mode=2

# deepstream-yolo parser entrypoint. This symbol name must match the built
# libnvdsinfer_custom_impl_Yolo.so (current marcoslucianops repo exports
# NvDsInferParseYolo; older revisions used NvDsInferParseCustomYoloV8).
parse-bbox-func-name=NvDsInferParseYolo
# The parser is named by ABSOLUTE path, NOT relative to this file: compose
# bind-mounts the host's deepstream/ over this directory, so a relative name
# would resolve to the HOST copy — which on a JetPack 5 board (DeepStream 6.3 /
# CUDA 11.4) is the wrong ABI and fails to load. /opt/vision-service/lib/ is
# outside the mount, so each image supplies the build matching its own
# DeepStream (see Dockerfile.deepstream and Dockerfile.deepstream.xavier).
custom-lib-path=/opt/vision-service/lib/libnvdsinfer_custom_impl_Yolo.so

# input-dims / output-blob-names are taken from the ONNX — do not pin them here.

# Preserve aspect ratio by letterboxing — prevents horizontal FoV distortion.
maintain-aspect-ratio=1
symmetric-padding=1

[class-attrs-all]
# Minimum confidence for nvinfer to pass a bbox to nvtracker. Keep low here;
# the pipeline's stabilizer and confidence gate filter downstream.
pre-cluster-threshold=0.1
nms-iou-threshold=0.45
topk=300
